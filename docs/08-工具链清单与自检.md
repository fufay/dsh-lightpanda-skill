## 3. brew 家底速查 / 补装


```sh
$BREW/bin/brew list --versions | wc -l   # 已装 formula 数（2026-10-24：134）
ls $BREW/bin | wc -l                      # 已链接命令数（665）
$BREW/bin/brew search <关键词>            # 查本仓有没有（源是 atomgit Harmonybrew，不是 homebrew-core）
$BREW/bin/brew install <formula>          # 装；装完把版本补进 `docs/00` §1 与本文 §4 自检清单
```

- **本仓提供但未装**（按需）：`ninja` `meson` `ccache` `bison` `flex` `nasm` `automake` `doxygen` `jq` `wget` `fd` `bat` `tree` `tmux` `shellcheck` `shfmt` `gnu-sed` `gawk`
- **本仓没有**：`gdb`（仅前端 `cgdb`）、`valgrind`、`graphviz`、`neovim`
- **系统自带（非 brew）**：`vim` 9.0、`zsh`、`zip`/`tar`/`file`、`unzip`（`$HOME/bin/unzip` = BusyBox 1.37.0）、`wget`（toybox 0.8.12）

## 4. 环境自检（一条命令）


```sh
BREW=/storage/Users/currentUser/.harmonybrew
export PATH=$BREW/opt/node@24/bin:$BREW/bin:/storage/Users/currentUser/npm/bin:/storage/Users/currentUser/Documents/cangjie/bin:$PATH

# 语言 / 包管理
for c in node npm go rustc cargo php java javac jar perl bash rg git uv python3 cjc devecocli agc; do
  printf '%-10s ' "$c"; command -v "$c" >/dev/null 2>&1 && echo ok || echo MISSING
done
# 编译 / 构建 + 鸿蒙打包签名
for c in clang clang++ gcc g++ cc make cmake ctest pkg-config patchelf autoconf libtool \
         llvm-ar llvm-readelf llvm-strip ld.lld lldb openssl curl sqlite3 ffmpeg zstd \
         hdc binary-sign-tool hap-sign-tool; do
  printf '%-16s ' "$c"; command -v "$c" >/dev/null 2>&1 && echo ok || echo MISSING
done

# 联网能力自检（brew python 自带 CA，直连 HTTPS 应 200）
# env -u PYTHONHOME：若从 Whale 内嵌 Python 派生的子进程里跑，继承的 PYTHONHOME 会让 brew python 起不来
env -u PYTHONHOME $BREW/bin/python3 -c "import urllib.request;print('brew-python HTTPS:',urllib.request.urlopen('https://www.baidu.com',timeout=10).status)"

# brew 家底（已装 formula 数 / 已链接命令数）
echo "formulae: $($BREW/bin/brew list --versions | wc -l)  bin: $(ls $BREW/bin | wc -l)"

# 可选：VM 兜底通道（需融合开发引擎运行；超时 = VM 没开）
VM=/storage/Users/currentUser/projects/DSH_Proj/certificate/lemo-vm
ssh -i "$VM/id_ed25519" -o IdentitiesOnly=yes -o StrictHostKeyChecking=accept-new \
    -o UserKnownHostsFile="$VM/known_hosts" -o ConnectTimeout=8 fox@172.16.105.2 \
    'uname -m; command -v dnf gcc; sudo -n true && echo sudo-ok'
```
